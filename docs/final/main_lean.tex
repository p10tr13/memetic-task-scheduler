\section{Problem}
% =====================================================================
Harmonogramujemy zadania na $M$ identycznych procesorach z~więzami pierwszeństwa. Problem jest
NP-trudny~\cite{ullman1975}, więc używamy metaheurystyki.

\noindent Instancja to czwórka $(T, p, l, M)$:
\begin{itemize}[itemsep=2pt, topsep=3pt]
    \item $T$ -- zbiór zadań;
    \item $p$ -- relacja pierwszeństwa (graf skierowany acykliczny, DAG): $t_i \to t_j$ znaczy, że $t_i$ kończy się przed startem $t_j$;
    \item $l$ -- czas trwania zadania;
    \item $M$ -- liczba procesorów.
\end{itemize}
\noindent Cel: zminimalizować \emph{makespan} $C_{max}$ -- czas zakończenia ostatniego zadania.
Rysunek~\ref{fig:przyklad} pokazuje przykład: graf zależności i~jego harmonogram.

\section{Metoda}
\label{sec:metoda}
% =====================================================================
Stosujemy algorytm memetyczny~\cite{moscato1989}: algorytm ewolucyjny (AE) z~lokalnym
przeszukiwaniem opartym o~ścieżkę krytyczną (CPM).

\paragraph{Reprezentacja i dekoder.} Osobnik to permutacja zadań -- kolejność priorytetów.
Zachłanny dekoder \emph{list scheduling}~\cite{graham1969} przydziela każde gotowe zadanie
procesorowi, który zwalnia się najwcześniej. Daje to harmonogram i~wartość $C_{max}$ (funkcja celu).

\paragraph{Naprawa.} Krzyżowanie i~mutacja mogą złamać pierwszeństwo. Naprawa listą zadań gotowych
przebudowuje permutację do najbliższego porządku topologicznego, więc cała populacja jest
dopuszczalna i~nie trzeba kar w~funkcji celu.

\paragraph{Operatory.} Selekcja turniejowa~\cite{miller1995}. Krzyżowanie permutacyjne: porządkowe
OX~\cite{davis1985} albo z~częściowym mapowaniem PMX~\cite{goldberg1985}. Mutacja: zamiana
(\emph{swap}) dwóch zadań.

\paragraph{CPM (świadomy harmonogramu).} Ścieżkę krytyczną wyznaczamy na \emph{zdekodowanym
harmonogramie}, wyjaśnienie na rys.~\ref{fig:przyklad}. Łapie to też wąskie gardła zasobowe. Zadania krytyczne
przesuwamy w~lewo (\emph{pull-left}); poprawę $C_{max}$ zapisujemy w~osobniku (ewolucja
Lamarckowska~\cite{whitley1994}).

\paragraph{Konfiguracje.} Dwa krzyżowania $\times$ CPM wł./wył. dają cztery warianty:
\texttt{ox}, \texttt{pmx} (czysty AE) oraz \texttt{ox\_cpm}, \texttt{pmx\_cpm} (memetyczny).

% =====================================================================
\section{Plan eksperymentów}
\label{sec:eksperymenty}
% =====================================================================
Sprawdzamy dwie hipotezy:
\begin{itemize}[itemsep=1pt, topsep=2pt, leftmargin=3em]
    \item[(H1)] CPM istotnie obniża medianę $C_{max}$ względem czystego AE;
    \item[(H2)] OX istotnie wygrywa z~PMX na grafach rzadkich ($|E|/|T|^2 < 0{,}1$).
\end{itemize}

\noindent\textbf{Dane.} 10 grafów rzeczywistych (Standard Task Graph~\cite{tobita2002}) i~10
syntetycznych o~kontrolowanej gęstości. Odniesienie: optimum MILP dla $|T|\le 30$, w.p.p. dolne
ograniczenie $LB=\max(\lceil \text{praca}/M\rceil,\ \text{ścieżka krytyczna})$.

\noindent\textbf{Protokół.} 15 niezależnych uruchomień na konfigurację (ziarna $0$--$14$); miara to
najlepszy $C_{max}$, tabele podają medianę. Istotność: sparowany test Wilcoxona~\cite{wilcoxon1945}
z~korekcją Holma, $\alpha=0{,}05$. Baseline: heurystyki HLFET~\cite{adam1974} i~MCP~\cite{wu1990}.

\noindent\textbf{Hiperparametry (globalne).} populacja $50$, pokolenia $100$, $p_c=0{,}8$,
$p_m=0{,}3$ (na osobnika), turniej $3$, prób CPM na osobnika $20$.

\noindent\textbf{Zastrzeżenie (H1).} Warianty memetyczne robią dodatkowe ewaluacje wewnątrz CPM,
więc porównujemy przy równej liczbie \emph{pokoleń}, nie ewaluacji.

\noindent 15 ziaren to mniej niż zwyczajowe 30 -- ograniczona moc statystyczna, podyktowana czasem
obliczeń. Ablacji hiperparametrów nie wykonano.

\section{Wnioski}
\label{sec:wnioski}
% =====================================================================
\begin{itemize}[itemsep=3pt, topsep=3pt]
    \item \textbf{H1 potwierdzona:} CPM istotnie obniża $C_{max}$ (25/25 testów), przy zastrzeżeniu
    o~równej liczbie pokoleń.
    \item \textbf{H2 odrzucona:} to PMX, nie OX, dominuje na grafach rzadkich (0/14 za OX).
    \item Najlepsza konfiguracja to \texttt{pmx\_cpm} -- najlepsza na każdej z~20 instancji, lepsza od
    HLFET/MCP na 19 z~nich. Największa luka do LB: \texttt{robot\_t88} (12,2\%).
    \item \textbf{Ograniczenia:} brak elityzmu; porównanie przy równej liczbie pokoleń (nie ewaluacji);
    15 ziaren; brak ablacji hiperparametrów.
    \item \textbf{Dalej:} ablacje hiperparametrów, elityzm, porównanie przy równym budżecie ewaluacji,
    więcej ziaren.
\end{itemize}
